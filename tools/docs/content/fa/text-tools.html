Normalizer

RtlyKit\Text\Normalizer متن فارسی را قابل مقایسه و ذخیره می‌کند. ورودی کاربر شکل‌های عربی و فارسی حروف را قاطی می‌کند (صفحه‌کلید عربی «ي» و «ك» تایپ می‌کند)، اعراب و کشیده و فاصلهٔ اضافه دارد؛ آن‌گاه یک واژه روی صفحه یکسان دیده می‌شود اما در مقایسهٔ رشته‌ای یا جست‌وجوی پایگاه داده نمی‌خواند.

normalize()

Normalizer::normalize($text, $removeDiacritics = true) این کارها را می‌کند:

  • حروف عربی به معادل فارسی تبدیل می‌شود: «ك» به «ک»، «ي» «ى» «ے» به «ی»، «ة» «ۀ» به «ه»، «ؤ» به «و»، «إ» «أ» «ٱ» به «ا».
  • ارقام عربی-هندی (٠-٩) به ارقام فارسی (۰-۹) تبدیل می‌شود. ارقام انگلیسی دست‌نخورده می‌ماند.
  • کشیده (تطویل، U+0640) حذف می‌شود.
  • اعراب (U+064B تا U+065F و U+0670) حذف می‌شود، مگر اینکه آرگومان دوم را false بدهید.
  • دنبالهٔ فاصله‌ها (فاصله، تب، خط جدید) به یک فاصله تبدیل و دو سر متن بریده می‌شود.
<?php
require 'vendor/autoload.php';

use RtlyKit\Text\Normalizer;
use function RtlyKit\normalize_text;

echo Normalizer::normalize("علي  كتاب  ١٢٣ 456"), "\n";    // علی کتاب ۱۲۳ 456
echo Normalizer::normalize("كــتاب"), "\n";                 // کتاب
echo Normalizer::normalize("  سلام \n\t دنیا  "), "\n";     // سلام دنیا
echo Normalizer::normalize("أحمد إبراهيم مؤمن ة"), "\n";    // احمد ابراهیم مومن ه
echo Normalizer::normalize("ثَبْت  ١٢٣ 456"), "\n";         // ثبت ۱۲۳ 456
echo Normalizer::normalize("ثَبْت", false), "\n";           // ثَبْت (اعراب حفظ شده)
echo normalize_text('كتاب'), "\n";                          // کتاب
به چه دست نمی‌زند. همزهٔ مستقل «ء» و «ئ» عمداً دست‌نخورده می‌ماند، چون در واژه‌های فارسی و عربی معنادار است. نیم‌فاصله (ZWNJ) حفظ می‌شود: «می‌خواهم» بایت‌به‌بایت همان می‌ماند. حروف انگلیسی تغییر نمی‌کند.

تابع کمکی normalize_text() همان Normalizer::normalize() با مقدارهای پیش‌فرض است.

fixHalfSpace()

نیم‌فاصله (ZWNJ، U+200C) چیزی است که «می‌خواهم» را درست نشان می‌دهد. کپی و پیست اغلب چند نیم‌فاصلهٔ پشت‌سرهم یا یکی کنار فاصله یا در انتهای متن جا می‌گذارد. fixHalfSpace() این‌ها را مرتب می‌کند:

  • خط تیرهٔ نرم (U+00AD) حذف می‌شود.
  • دنبالهٔ نیم‌فاصله‌ها به یکی تبدیل می‌شود.
  • نیم‌فاصلهٔ چسبیده به فاصله (قبل یا بعد) حذف می‌شود.
  • نیم‌فاصلهٔ ابتدا و انتهای متن حذف می‌شود.
$a = "می\u{200C}\u{200C}\u{200C}خواهم";
echo mb_strlen($a), ' ', mb_strlen(Normalizer::fixHalfSpace($a)), "\n";   // 10 8

$b = "\u{200C}سلام \u{200C}دنیا\u{200C}";
echo mb_strlen($b), ' ', mb_strlen(Normalizer::fixHalfSpace($b)), "\n";   // 12 9
echo Normalizer::fixHalfSpace($b), "\n";                                  // سلام دنیا

نیم‌فاصلهٔ جاافتاده را اضافه نمی‌کند و فاصله‌های معمولی را هم یکی نمی‌کند.

clean()

Normalizer::clean() نسخهٔ همه‌کاره برای ذخیره و جست‌وجوست: ابتدا normalize()، سپس fixHalfSpace() و بعد حذف نویسه‌های نامرئی فاصلهٔ با پهنای صفر (U+200B)، اتصال‌دهندهٔ با پهنای صفر (U+200D) و نشانهٔ ترتیب بایت (U+FEFF). ZWNJ تنها نویسهٔ با پهنای صفری است که می‌ماند.

$c = "كتاب\u{200B}ي  ١٢٣";
echo Normalizer::clean($c), "\n";   // کتابی ۱۲۳

اگر فقط normalize() را روی همین متن بزنید، فاصلهٔ با پهنای صفر نامرئی درون واژه می‌ماند؛ دلیل وجود clean() همین است. پیش از ذخیره یا نمایه‌سازی متن واردشده توسط کاربر از آن استفاده کنید.

Detector

RtlyKit\Text\Detector دربارهٔ خط، زبان و جهت پاسخ می‌دهد. همهٔ متدها ایستا هستند و هرگز استثنا نمی‌اندازند.

isPersian() و isArabic()

هر دو روش ابتکاری برای متن با خط عربی‌اند. فارسی از روی حروفی شناخته می‌شود که فقط در فارسی هست (پ چ ژ گ، شکل‌های فارسی ک و ی و ارقام فارسی). عربی از ي ك ى ة و ارقام عربی-هندی شناخته می‌شود. حروف مشترک، از جمله «ه» و صورت‌های همزه، شمرده نمی‌شوند. اگر حروف ویژهٔ فارسی دست‌کم به اندازهٔ حروف ویژهٔ عربی باشند متن فارسی است؛ پس تساوی و متنِ بدون حرف تمایزدهنده فارسی حساب می‌شود.

use RtlyKit\Text\Detector;

var_dump(Detector::isPersian('سلام'));        // bool(true)، حرف تمایزدهنده ندارد، پیش‌فرض فارسی
var_dump(Detector::isPersian('گچپژ'));        // bool(true)
var_dump(Detector::isArabic('مرحبا بكم'));    // bool(true)
var_dump(Detector::isArabic('كتاب ي'));       // bool(true)
var_dump(Detector::isPersian('hello'));       // bool(false)، اصلاً خط عربی ندارد
var_dump(Detector::isPersian('۱۲۳'));         // bool(true)، ارقام فارسی
var_dump(Detector::isArabic('٣٤٥'));          // bool(true)، ارقام عربی-هندی
var_dump(Detector::isPersian(''));            // bool(false)
محدودیت شناخته‌شده. این روش بر بسامد حروف بنا شده و تشخیص‌دهندهٔ زبان نیست. متن بسیار کوتاه، نام‌ها و واژه‌های عربی که فارسی‌نویسی شده‌اند مبهم‌اند و متن آمیخته می‌تواند به هر دو سو برود. برای تصمیمی که باید حتماً درست باشد از آن استفاده نکنید؛ برای انتخاب یک پیش‌فرض معقول به کار می‌آید.

جهت و RTL

  • containsRtl($text): آیا متن نویسه‌ای از یک خط راست‌به‌چپ (عبری، عربی و ضمیمه‌هایش، سریانی، تانا، نکو و غیره، شکل‌های ارائهٔ عربی) یا نشانهٔ RLM دارد.
  • direction($text): 'rtl' یا 'ltr' بر پایهٔ نخستین حرف قوی طبق قاعدهٔ دوسویهٔ یونیکد. رقم، نشانه‌گذاری و فاصله خنثی‌اند. متنِ بدون حرف 'ltr' برمی‌گرداند.
  • isHebrew($text): آیا متن حرف عبری دارد.
var_dump(Detector::containsRtl('abc'));           // bool(false)
echo Detector::direction('سلام hello'), "\n";    // rtl
echo Detector::direction('Hello مرحبا'), "\n";   // ltr (اولین حرف لاتین است)
echo Detector::direction('۱۲۳'), "\n";           // ltr (رقم خنثی است و حرفی نیست)
var_dump(Detector::isHebrew('שלום'));            // bool(true)

توابع کمکی contains_rtl() و text_direction() به‌ترتیب containsRtl() و direction() را پوشش می‌دهند. برای صفت dir در HTML روی متن تولیدشدهٔ کاربر از direction() استفاده کنید.

isRtlLocale()

isRtlLocale($locale) می‌گوید یک برچسب زبان راست‌به‌چپ نوشته می‌شود یا نه. بر پایهٔ -، _، . و @ می‌شکند، به بزرگی و کوچکی حروف و فاصله‌های دو سر توجه ندارد و اگر زبان راست‌به‌چپ باشد (fa، ar، he، ur، ps، ug، dv، ckb، azb و غیره) یا زیربرچسب خطی مثل Arab یا Hebr باشد، true می‌دهد.

var_dump(Detector::isRtlLocale('fa_IR'));     // bool(true)
var_dump(Detector::isRtlLocale('ar-SA'));     // bool(true)
var_dump(Detector::isRtlLocale('ckb-IQ'));    // bool(true)
var_dump(Detector::isRtlLocale('az-Arab'));   // bool(true)، زیربرچسب خط تعیین‌کننده است
var_dump(Detector::isRtlLocale('ku'));        // bool(false)، «ku» ساده در فهرست زبان‌های RTL نیست
var_dump(Detector::isRtlLocale('en'));        // bool(false)
var_dump(Detector::isRtlLocale(''));          // bool(false)

Slugify

Slugify::make($text, $separator = '-') عنوان را به اسلاگ مناسب نشانی تبدیل می‌کند و حروف فارسی و عربی را خوانا نگه می‌دارد. هرگز نویسه‌گردانی نمی‌کند.

  1. متن با Normalizer::normalize() نرمال می‌شود.
  2. فاصله و نیم‌فاصله به جداکننده تبدیل می‌شود.
  3. هر چیز غیر از حرف (هر خطی)، نشانهٔ ترکیبی، رقم (هر دسته‌ای)، خط تیره و زیرخط حذف می‌شود. نشانه‌گذاری و اموجی ناپدید می‌شود.
  4. خط تیره، زیرخط و جداکنندهٔ تکراری یکی می‌شود و جداکننده از دو سر بریده می‌شود.
  5. نتیجه به صورت UTF-8 به حروف کوچک تبدیل می‌شود.
use RtlyKit\Text\Slugify;

echo Slugify::make('سلام دنیا'), "\n";               // سلام-دنیا
echo Slugify::make("کتاب\u{200C}خانه ملی"), "\n";    // کتاب-خانه-ملی
echo Slugify::make('Hello, World!'), "\n";           // hello-world
echo Slugify::make('  A--B__C  '), "\n";             // a-b_c
echo Slugify::make('سلام!!! دنیا؟'), "\n";           // سلام-دنیا
echo Slugify::make('۱۲۳ ابر'), "\n";                 // ۱۲۳-ابر
echo Slugify::make('كتاب ي'), "\n";                  // کتاب-ی
echo Slugify::make('سلام 😀 دنیا'), "\n";            // سلام-دنیا
echo Slugify::make('عکس.jpg'), "\n";                 // عکسjpg
var_dump(Slugify::make('$%^&'));                    // string(0) ""

دو نکته: ارقام همان دسته‌ای می‌مانند که تایپ شده (۱۲۳ فارسی می‌ماند)؛ اگر ارقام ASCII در نشانی می‌خواهید ابتدا Digits::toEnglish() را اجرا کنید. و نقطه حذف می‌شود، نه جداکننده؛ پس پسوند فایل به نام می‌چسبد (عکسjpg) و نام و پسوند را جدا اسلاگ کنید.

جداکننده

هر رشته‌ای را می‌توانید آرگومان دوم بدهید: '_'، رشتهٔ بلندتر یا رشتهٔ خالی برای چسباندن واژه‌ها بدون جداکننده.

echo Slugify::make('سلام دنیا', '_'), "\n";        // سلام_دنیا
echo Slugify::make('سلام دنیا', ''), "\n";         // سلامدنیا
echo Slugify::make('Hello World', '--'), "\n";     // hello--world

جداکننده حداکثر ۶۴ بایت و UTF-8 معتبر باشد. جداکنندهٔ بلندتر RtlyKitException با کد input_too_long می‌اندازد و جداکنندهٔ نامعتبر همان استثنا را با invalid_argument:

try {
    Slugify::make('Hello World', str_repeat('x', 65));
} catch (\RtlyKit\Exceptions\RtlyKitException $e) {
    echo $e->getMessage(), ' [', $e->getErrorCode()->value, "]\n";
    // The slug separator is too long. [input_too_long]
}
اسلاگ خالی و یکتایی. عنوانی که فقط نشانه‌گذاری باشد رشتهٔ خالی می‌دهد و عنوان‌های متفاوت می‌توانند اسلاگ یکسان بسازند. حالت خالی را مدیریت کنید (مثلاً به شناسه برگردید) و یکتایی را در لایهٔ ذخیره‌سازی خودتان تضمین کنید.
کدگذاری. Slugify متن UTF-8 را انتظار دارد. خود متن را اعتبارسنجی نمی‌کند و بایت‌های نامعتبر ورودی گزارش نمی‌شوند. ورودی نامطمئن را پیش از آن بررسی یا تبدیل کنید.
